The repo

A small pattern matching static analyzer for smart contract source files. It reads one contract file as text and reports a short list of known risky patterns. It does not build a control flow graph, parse Solidity, or prove that a contract is safe or exploitable.

def main():
    """
    Main function to execute the smart contract static analyzer.
    """
    parser = setup_argparse()
    args = parser.parse_args()

    if args.verbose:
        logging.getLogger().setLevel(logging.DEBUG)  # Set logging to debug level

    contract_code = validate_input(args.contract_code)
    if not analyze_contract(contract_code, args.output):
        sys.exit(1)

# Example usage:
# python main.py contract.sol --output analysis.txt
# python main.py contract.bin -v

if __name__ == "__main__":
    main()
